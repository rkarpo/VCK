function submitVote() { const nameInput = document.getElementById('memberName').value.trim(); const submitBtn = document.querySelector('.btn-submit'); if (!nameInput) { alert('Bitte gib deinen Namen ein!'); return; } if (!selectedQ1) { alert('Bitte antworte auf Frage 1 (Format des Turniers)!'); return; } if (!selectedQ2) { alert('Bitte antworte auf Frage 2 (Teilnahme als Spieler)!'); return; } // Button deaktivieren, um Mehrfach-Klicks zu verhindern submitBtn.disabled = true; submitBtn.innerText = "Wird gespeichert..."; votesRef.push({ name: nameInput, vote_q1: selectedQ1, vote_q2: selectedQ2, timestamp: Date.now() }).then(() => { alert('Vielen Dank! Deine Stimme wurde erfolgreich gespeichert.'); // Formular zurücksetzen document.getElementById('memberName').value = ''; selectedQ1 = null; selectedQ2 = null; document.querySelectorAll('.option-btn').forEach(btn => btn.classList.remove('selected')); document.querySelectorAll('input[type="radio"]').forEach(radio => radio.checked = false); }).catch(error => { alert("Fehler beim Speichern in Firebase: " + error.message); }).finally(() => { // Button wieder aktivieren submitBtn.disabled = false; submitBtn.innerText = "Abstimmen"; }); }
